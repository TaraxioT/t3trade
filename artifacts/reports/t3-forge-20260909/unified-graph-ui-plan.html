export function timeFraction(atMs: number, fromMs: number, toMs: number): number | null {
  if (!Number.isFinite(atMs) || !Number.isFinite(fromMs) ||
      !Number.isFinite(toMs) || toMs <= fromMs || atMs < fromMs || atMs > toMs) {
    return null;
  }
  return (atMs - fromMs) / (toMs - fromMs);
}
